1 Create a Free Supabase Project & Run This SQL

Create a free project at supabase.com, disable Confirm email under Authentication → Providers → Email (for instant sign-up), then paste this SQL in the SQL Editor and click Run: 

-- 1. Gatherings Table (Public board + archive)
create table if not exists gatherings (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  description text not null,
  date_time timestamptz not null,
  venue text not null,
  organizer_id uuid references auth.users not null,
  organizer_name text not null,
  max_capacity int not null default 8,
  is_closed boolean not null default false,
  is_archived boolean not null default false,
  archived_at timestamptz,
  created_at timestamptz default now()
);

-- 2. Participants Table (Emails protected by Row-Level Security!)
create table if not exists gathering_participants (
  id uuid default gen_random_uuid() primary key,
  gathering_id uuid references gatherings(id) on delete cascade not null,
  user_id uuid references auth.users not null,
  display_name text not null,
  email text not null,
  joined_at timestamptz default now(),
  unique(gathering_id, user_id)
);

-- 3. Public View (Exposes names ONLY; strips all emails from public)
create or replace view public_gathering_participants as
  select id, gathering_id, user_id, display_name, joined_at
  from gathering_participants;

grant select on public_gathering_participants to anon, authenticated;

-- 4. Enable Row-Level Security (RLS)
alter table gatherings enable row level security;
alter table gathering_participants enable row level security;

create policy "Anyone can read gatherings" on gatherings for select using (true);
create policy "Auth users create gatherings" on gatherings for insert with check (auth.uid() = organizer_id);
create policy "Organizers update own gatherings" on gatherings for update using (auth.uid() = organizer_id);

-- Only organizer (or participant themselves) can read raw table with emails!
create policy "Organizer or self reads emails" on gathering_participants
  for select using (
    auth.uid() = user_id or
    exists (select 1 from gatherings g where g.id = gathering_id and g.organizer_id = auth.uid())
  );
create policy "Auth users join open gatherings" on gathering_participants
  for insert with check (auth.uid() = user_id);
create policy "Participants can leave" on gathering_participants
  for delete using (auth.uid() = user_id);
